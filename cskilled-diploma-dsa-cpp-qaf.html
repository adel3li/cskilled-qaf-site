binary_search.cpp
int binary_search(const vector<int>& items, int target) {
    int low = 0, high = (int)items.size() - 1;
    while (low <= high) {
        int mid = low + (high - low) / 2;
        if (items[mid] == target) return mid;
        if (items[mid] < target) low = mid + 1;
        else high = mid - 1;
    }
    return -1;   // O(log n)
}
